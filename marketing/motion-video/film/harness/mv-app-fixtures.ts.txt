const pricing = catalogFallbackPricing();
export const REQUESTS = PEOPLE.map(([leadName, leadCompany, daysAgo, status, tweak], i) => {

function overview() {
  const inWindow = REQUESTS.filter((r) => r._creationTime >= NOW - 30 * DAY);
  const byStatus: Record<string, number> = {};
  let wonValueCents = 0, quoted = 0;
  for (const r of inWindow) {
    byStatus[r.status] = (byStatus[r.status] ?? 0) + 1;
    if (r.status === "won") wonValueCents += r.priceCents;
    quoted += r.priceCents;
  }
  const won = byStatus.won ?? 0;
  const widgetViews = 1240;
  const trend = Array.from({ length: 30 }, (_, k) => {
    const end = NOW - (29 - k) * DAY, start = end - DAY;
    const rows = inWindow.filter((r) => r._creationTime >= start && r._creationTime < end);
    const d = new Date(end);
    return { label: `${d.getUTCDate()}/${d.getUTCMonth() + 1}`, count: rows.length, valueCents: rows.reduce((s, r) => s + r.priceCents, 0) };
  });
  return {
    range: "1m", analyticsLevel: "full",
    previous: { totalRequests: 14, won: 4, wonValueCents: Math.round(wonValueCents * 0.72), conversionRate: 4 / 14, widgetViews: 980, visitorConversionRate: 14 / 980, avgDealCents: Math.round(wonValueCents / Math.max(1, won) * 0.9), pipelineValueCents: Math.round(quoted * 0.7) },
    totalRequests: inWindow.length, widgetViews, widgetViewsApprox: false, visitorConversionRate: inWindow.length / widgetViews,
    realLeads: inWindow.length, won, lost: byStatus.lost ?? 0, spam: 0, conversionRate: won / inWindow.length,
    wonValueCents, pipelineValueCents: quoted, avgDealCents: won ? Math.round(wonValueCents / won) : 0, byStatus,
    funnel: [{ key: "new", count: byStatus.new ?? 0 }, { key: "contacted", count: byStatus.contacted ?? 0 }, { key: "quoted", count: byStatus.quoted ?? 0 }, { key: "won", count: won }],
    trend, byConfigurator: [{ name: "Configuratore sito web", count: inWindow.length - 4, valueCents: Math.round(quoted * 0.8) }, { name: "Pagina social", count: 4, valueCents: Math.round(quoted * 0.2) }],
    bySource: [{ host: "www.serramentidemo.it", count: inWindow.length - 5 }, { host: "instagram.com", count: 3 }, { host: "diretto / sconosciuto", count: 2 }],
    truncated: false,
  };
}
