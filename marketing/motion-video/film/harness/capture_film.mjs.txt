a[href*="tab=plan"],a[href*="tab=billing"]{display:none!important}`;
const box = async (loc) => { const r = await loc.boundingBox(); return r && { x: r.x, y: r.y, w: r.width, h: r.height }; };

async function appPage(view, navHref, fn) {
  const c = await b.newContext({ viewport: { width: 1920, height: 1080 }, deviceScaleFactor: 2, ignoreHTTPSErrors: true });
  const p = await c.newPage();
  await p.goto(`http://localhost:3100/c/mv-app/${view}`, { waitUntil: "networkidle", timeout: 120000 });
  await p.addStyleTag({ content: HIDE });
  // topbar plan badge ("Pro") — hide: it links to the plan page
  await p.evaluate((href) => {
    for (const a of document.querySelectorAll("header a, header button")) if (/^\s*Pro\s*$/.test(a.textContent || "")) a.style.display = "none";
    const link = [...document.querySelectorAll("#app-sidebar-nav a")].find((a) => a.getAttribute("href")?.endsWith(href));
    if (link) { link.className = link.className.replace("text-[var(--color-text-secondary)]", "").replace(/hover:\S+/g, "") + " bg-[var(--color-mint)]/15 text-[var(--color-mint)]"; }
  }, navHref);
  await p.waitForTimeout(1500);
  await fn(p);
  await c.close();
}
